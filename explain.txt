WORKDIR /app/backend,Home Directory: This is where all your app's code will live inside the container. All the next commands will run from this /app/backend folder.
RUN apt-get update && apt-get install -y --no-install-recommends \ ...,"System Tools: Bohot lambi line hai, but simple hai! This installs all the Linux system dependencies your Python code (and especially tools like wkhtmltopdf for PDF generation) needs. Things like gcc (for compiling C code), graphics libs (libcairo2-dev), and fonts (fonts-dejavu-core) are all included. Finally, it cleans up all the temporary download files (rm -rf /var/lib/apt/lists/*) to keep the image small."
RUN wget https://github.com/wkhtmltopdf/...,"PDF Generator Fix: This downloads and installs a specific version of wkhtmltopdf. This tool converts HTML to PDF. The special part? This version doesn't have the headache of the old libssl1.1 dependency, making it much easier to run in this slim base image. It downloads, installs (dpkg -i), fixes any missing dependencies (apt-get install -f -y), and then deletes the downloaded file (rm ...deb)."
RUN python -m venv /app/backend/venv,Virtual Environment: This creates a dedicated space for your Python packages called a venv. It's good practice so your project's libraries don't clash with the system's libraries.
RUN /app/backend/venv/bin/pip install --upgrade pip,Update Pip: Just updating the Python package manager (pip) inside the venv to the newest version. Standard operating procedure.
COPY reqs.txt .,Copy Files: Takes your reqs.txt file (where all your project's required Python libraries are listed) from your local computer and puts it into the container.
RUN /app/backend/venv/bin/pip install --no-cache-dir -r reqs.txt && \ /app/backend/venv/bin/pip install --no-cache-dir pandas,Install Libraries: This uses the venv's pip to install everything listed in reqs.txt. The --no-cache-dir keeps the image clean. And they are installing pandas separately too (maybe it's a critical dependency they want to highlight).
COPY . .,"Copy Code: Now that all dependencies are installed, this copies all the remaining files from your project folder into the container's working directory (/app/backend)."
"ENV PATH=""/app/backend/venv/bin:$PATH""","Set Path: This is important! It makes sure that when the container starts, any command (like python or gunicorn) runs using the programs inside your virtual environment, not the system-wide Python."
ENV DJANGO_SETTINGS_MODULE=tars.settings,"Django Setup: Tells your Django app which settings file to use (tars.settings), which is crucial for connection details, database, etc."
COPY start.sh /app/backend/start.sh,"Copy Scripts: Copies the start.sh script, which probably contains the actual command to run the web server."
COPY setup.sh /app/backend/setup.sh,"Copies the setup.sh script, which likely contains commands to run migrations, collect static files, and maybe create a superuser before starting the server."
EXPOSE 8000 8001,Open Doors: This is just documentation. It tells other developers that this container will listen on ports 8000 and 8001.
"CMD [""/app/backend/setup.sh""]","The Launch Command: When the container finally starts running, it will execute the setup.sh script. This script will handle initial setup steps and then probably call start.sh to run the web server."
